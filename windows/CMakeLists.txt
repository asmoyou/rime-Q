cmake_minimum_required(VERSION 3.21)
project(RimeQWindows VERSION 0.4.1 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /utf-8 /permissive- /EHsc)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
set(RIMEQ_BUILD_NUMBER "1" CACHE STRING "Monotonic Windows build number")
configure_file(resources/version.h.in generated/version.h @ONLY)
include_directories(common resources "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_library(rimeq_artwork STATIC visuals/render.cpp)
target_link_libraries(rimeq_artwork PUBLIC gdiplus user32 gdi32)
add_library(RimeQVisuals SHARED visuals/preview.cpp)
target_link_libraries(RimeQVisuals PRIVATE rimeq_artwork ole32 advapi32 shell32)
set_target_properties(RimeQVisuals PROPERTIES OUTPUT_NAME "RimeQ.Visuals")
option(RIMEQ_BUILD_TSF "Build the native text service" ON)
if(RIMEQ_BUILD_TSF)
  add_library(RimeQTip SHARED tsf/service.cpp tsf/candidate.cpp tsf/registration.cpp resources/tip.rc)
  target_link_libraries(RimeQTip PRIVATE rimeq_artwork ole32 oleaut32 uuid user32 gdi32 gdiplus advapi32 shell32 shlwapi dwmapi)
  set_target_properties(RimeQTip PROPERTIES OUTPUT_NAME "RimeQ.Tip")
endif()
add_executable(RimeQBroker broker/main.cpp broker/engine.cpp resources/broker.rc)
target_link_libraries(RimeQBroker PRIVATE ole32 advapi32 shell32 user32 bcrypt)
set_target_properties(RimeQBroker PROPERTIES OUTPUT_NAME "RimeQ.Broker")
add_executable(RimeQControl broker/control.cpp resources/broker.rc)
target_link_libraries(RimeQControl PRIVATE ole32 uuid advapi32 shell32)
set_target_properties(RimeQControl PROPERTIES OUTPUT_NAME "RimeQ.Control")
include(CTest)
if(BUILD_TESTING)
  add_executable(rimeq_windows_tests tests/protocol_tests.cpp)
  target_link_libraries(rimeq_windows_tests PRIVATE ole32 advapi32 shell32)
  add_test(NAME windows_protocol COMMAND rimeq_windows_tests)
  add_executable(rimeq_profile_tests tests/profile_tests.cpp)
  target_link_libraries(rimeq_profile_tests PRIVATE ole32 uuid advapi32)
  add_test(NAME windows_profile_lifecycle COMMAND rimeq_profile_tests)
  add_executable(rimeq_tsf_tests tests/tsf_tests.cpp broker/engine.cpp)
  target_link_libraries(rimeq_tsf_tests PRIVATE ole32 oleaut32 uuid advapi32 shell32 user32 bcrypt)
  add_executable(rimeq_candidate_tests tests/candidate_tests.cpp tsf/candidate.cpp)
  target_link_libraries(rimeq_candidate_tests PRIVATE rimeq_artwork ole32 advapi32 shell32 user32 gdi32 gdiplus dwmapi)
  add_executable(rimeq_host_tests tests/host_tests.cpp)
  target_link_libraries(rimeq_host_tests PRIVATE ole32 uuid advapi32 shell32 user32 version)
  add_executable(rimeq_sync_engine_tests tests/sync_engine_tests.cpp broker/engine.cpp)
  target_link_libraries(rimeq_sync_engine_tests PRIVATE ole32 advapi32 shell32 user32 bcrypt)
  add_executable(rimeq_sync_engine_node tests/sync_engine_node.cpp broker/engine.cpp)
  target_link_libraries(rimeq_sync_engine_node PRIVATE ole32 advapi32 shell32 user32 bcrypt)
endif()
